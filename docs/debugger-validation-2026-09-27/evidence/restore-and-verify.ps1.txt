$ErrorActionPreference = 'Stop'
$root = 'C:\Users\Administrator\Desktop\Projects\RocketIDE-Build\wp06-20260927'
$profile = [IO.Path]::GetFullPath('C:\Users\Administrator\AppData\Local\RocketIDE')
$backup = Join-Path $root 'profile-original'
$manifest = @(Get-Content -LiteralPath (Join-Path $root 'profile-original-manifest.json') -Raw | ConvertFrom-Json)
$processes = @(Get-CimInstance Win32_Process | Where-Object { $_.Name -in @('RocketIDE.exe','EngHost.exe','main.exe','rocket-lsp.exe') } | Select-Object ProcessId,ParentProcessId,ExecutablePath)
ConvertTo-Json -InputObject $processes | Set-Content -LiteralPath (Join-Path $root 'evidence\final-product-processes.json')
if ($processes.Count -ne 0) { throw 'Close product processes before restoration' }
foreach ($entry in $manifest) {
    $source = [IO.Path]::GetFullPath((Join-Path $backup $entry.File))
    if (-not $source.StartsWith($backup + '\', [StringComparison]::OrdinalIgnoreCase)) { throw 'Backup path outside root' }
    if ((Get-FileHash -LiteralPath $source).Hash -ne $entry.SHA256) { throw "Backup mismatch: $source" }
}
$testBackup = Join-Path $root 'profile-test-final'
if (Test-Path -LiteralPath $testBackup) { throw 'Do not overwrite saved test profile' }
Copy-Item -LiteralPath $profile -Destination $testBackup -Recurse
$expected = @{}
foreach ($entry in $manifest) { $expected[$entry.File] = $entry }
$removed = @()
foreach ($file in @(Get-ChildItem -LiteralPath $profile -File -Recurse)) {
    $resolved = [IO.Path]::GetFullPath($file.FullName)
    if (-not $resolved.StartsWith($profile + '\', [StringComparison]::OrdinalIgnoreCase)) { throw 'Profile path outside root' }
    $relative = [IO.Path]::GetRelativePath($profile, $resolved)
    if (-not $expected.ContainsKey($relative)) {
        $removed += $relative
        Remove-Item -LiteralPath $resolved -Force
    }
}
foreach ($entry in $manifest) {
    $destination = [IO.Path]::GetFullPath((Join-Path $profile $entry.File))
    if (-not $destination.StartsWith($profile + '\', [StringComparison]::OrdinalIgnoreCase)) { throw 'Restore path outside root' }
    New-Item -ItemType Directory -Path (Split-Path -Parent $destination) -Force | Out-Null
    Copy-Item -LiteralPath (Join-Path $backup $entry.File) -Destination $destination -Force
}
$files = @(Get-ChildItem -LiteralPath $profile -File -Recurse)
$mismatches = @($files | ForEach-Object {
    $relative = [IO.Path]::GetRelativePath($profile, $_.FullName)
    $original = $expected[$relative]
    if ($null -eq $original -or $_.Length -ne $original.Bytes -or (Get-FileHash -LiteralPath $_.FullName).Hash -ne $original.SHA256) { $relative }
})
if ($files.Count -ne $manifest.Count -or $mismatches.Count -ne 0) { throw 'Profile restoration verification failed' }
[pscustomobject]@{ ExpectedFiles=$manifest.Count; RestoredFiles=$files.Count; Mismatches=$mismatches; TestFilesRemoved=$removed; OriginalBackup=$backup; TestBackup=$testBackup } | ConvertTo-Json -Depth 8 | Tee-Object -FilePath (Join-Path $root 'evidence\profile-restoration.json')
$audit = @(Get-Content -LiteralPath (Join-Path $root 'evidence\audit-preservation-before.json') -Raw | ConvertFrom-Json)
$auditMismatch = @($audit | Where-Object { (Get-Item -LiteralPath $_.Path).Length -ne $_.Length -or (Get-FileHash -LiteralPath $_.Path).Hash -ne $_.SHA256 })
$fixtures = @(Get-Content -LiteralPath (Join-Path $root 'evidence\fixtures-before-final-replay.json') -Raw | ConvertFrom-Json)
$fixtureMismatch = @($fixtures | Where-Object { (Get-FileHash -LiteralPath $_.File).Hash -ne $_.SHA256 })
[pscustomobject]@{ AuditFiles=$audit.Count; AuditMismatches=$auditMismatch; FixtureFiles=$fixtures.Count; FixtureMismatches=$fixtureMismatch } | ConvertTo-Json -Depth 8 | Tee-Object -FilePath (Join-Path $root 'evidence\preservation-verification.json')
if ($auditMismatch.Count -ne 0 -or $fixtureMismatch.Count -ne 0) { throw 'Preservation mismatch' }
