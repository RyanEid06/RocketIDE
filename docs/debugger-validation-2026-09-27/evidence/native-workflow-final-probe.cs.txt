using System.Diagnostics;
using RocketIDE.Debugger;
var clock=Stopwatch.StartNew();
void Log(string text)=>Console.WriteLine($"{clock.Elapsed.TotalSeconds:F3}s {text}");
var root=args[0];
var bin=Path.Combine(root,".rocketc","targets","windows-x64");
var source=Path.Combine(root,"src","main.rocket");
var request=new RocketDebugLaunchRequest(Path.Combine(bin,"main.exe"),Path.Combine(bin,"main.pdb"),Path.Combine(bin,"main.rocket.map.json"),root,root,[],[new(source,9)]);
for(var iteration=0;iteration<2;iteration++)
{
 using var deadline=new CancellationTokenSource(TimeSpan.FromSeconds(20));
 var transport=await DbgXCommandTransport.CreateAsync(deadline.Token);
 await using var debugger=new RocketNativeDebugger(transport);
 debugger.StateChanged+=(_,e)=>Log($"STATE {e.State}: {e.Message}");
 await debugger.LaunchAsync(request,deadline.Token);
 if(debugger.State!=RocketDebugSessionState.Stopped || debugger.CurrentLocation?.Line!=9) throw new Exception("Expected source9 stop");
 Log("EVALUATE "+await debugger.EvaluateAsync("seed",deadline.Token));
 await debugger.RunToCursorAsync(source,10,deadline.Token);
 if(debugger.HasTemporaryBreakpoint || debugger.CurrentLocation?.Line!=10) throw new Exception("Temporary breakpoint cleanup/location failed");
 Log("AT CURSOR "+await debugger.EvaluateAsync("answer",deadline.Token));
 var breakpointTable=await transport.ExecuteAsync("bl",deadline.Token);
 Log("BREAKPOINT TABLE "+breakpointTable);
 if(System.Text.RegularExpressions.Regex.Matches(breakpointTable,@"(?m)^\s*\d+\s+[ed]\s+").Count!=1) throw new Exception("Unexpected hidden breakpoint after cursor hit");
 var running=debugger.ContinueAsync(deadline.Token);
 await Task.Delay(3000,deadline.Token);
 if(debugger.State!=RocketDebugSessionState.Running) throw new Exception("Valid long run ended early");
 await debugger.PauseAsync(deadline.Token);
 await running.WaitAsync(deadline.Token);
 await debugger.SelectThreadAsync(0,deadline.Token);
 Log("THREAD0 FRAMES "+debugger.Frames.Count);
 var pendingCursor=debugger.RunToCursorAsync(source,16,deadline.Token);
 await Task.Delay(300,deadline.Token);
 if(!debugger.HasTemporaryBreakpoint) throw new Exception("Expected pending temporary breakpoint");
 await debugger.PauseAsync(deadline.Token);
 await pendingCursor.WaitAsync(deadline.Token);
 breakpointTable=await transport.ExecuteAsync("bl",deadline.Token);
 Log("BREAKPOINT TABLE AFTER PAUSE "+breakpointTable);
 if(debugger.HasTemporaryBreakpoint || System.Text.RegularExpressions.Regex.Matches(breakpointTable,@"(?m)^\s*\d+\s+[ed]\s+").Count!=1) throw new Exception("Hidden breakpoint after cancellation by Pause");
 var beforeStop=clock.ElapsedMilliseconds;
 await debugger.StopAsync(deadline.Token);
 Log($"STOP completed in {clock.ElapsedMilliseconds-beforeStop}ms; temp={debugger.HasTemporaryBreakpoint}");
}
Log("NATIVE WORKFLOW PASS");
