using System.Diagnostics;
using RocketIDE.Debugger;
using DbgX.Interfaces.Services;

var clock = Stopwatch.StartNew();
void Log(string message) => Console.WriteLine($"{clock.Elapsed.TotalSeconds:F3}s {message}");
var options = new EngineOptions();
foreach (var property in typeof(EngineOptions).GetProperties())
    Log($"OPTION {property.PropertyType.FullName} {property.Name} = {property.GetValue(options)}");
if (args.Length == 0) return;
using var deadline = new CancellationTokenSource(TimeSpan.FromSeconds(30));
var transport = await DbgXCommandTransport.CreateAsync(deadline.Token);
transport.OutputReceived += (_, e) => Log("OUTPUT " + e.Text);
try
{
    Log("CREATE begin");
    await transport.CreateProcessAsync(args[0], [], Path.GetDirectoryName(args[0])!, deadline.Token);
    Log("CREATE end");
    var processText = await transport.ExecuteAsync("|", deadline.Token);
    var processId = DbgEngProtocol.ParseCurrentProcessId(processText)!.Value;
    foreach (var command in new[] { ".sympath C:\\Users\\Administrator\\Desktop\\Projects\\RocketIDE-Build\\wp06-20260927\\fixtures\\debug-proof\\.rocketc\\targets\\windows-x64", ".expr /s masm", ".lines -e", "l+t", ".srcpath+ \"C:\\Users\\Administrator\\Desktop\\Projects\\RocketIDE-Build\\wp06-20260927\\fixtures\\debug-proof\\src\"", "|", "ld \"main\"", "bc *", "bp `main.rocket:9`", "g", "|", "?? seed", "?? answer", "?? missing", "?? (", "t", "|", "p", "|", "dv /t", "?? value", ".frame 1", "dv /t", "?? seed", "~0s", ".frame 0", "dv /t" })
    {
        Log("COMMAND " + command);
        Log(await transport.ExecuteAsync(command, deadline.Token));
    }
    Log("CONTINUE for running evaluation probe");
    await transport.ExecuteAsync("g", deadline.Token);
    await Task.Delay(200);
    using (var evaluationDeadline = new CancellationTokenSource(TimeSpan.FromMilliseconds(250)))
    {
        try { Log("RUNNING EVAL " + await transport.ExecuteAsync("?? value", evaluationDeadline.Token)); }
        catch (OperationCanceledException) { Log("RUNNING EVAL unavailable: queued until stop, deadline expired"); }
    }
    Log("PAUSE begin");
    await transport.BreakAsync(processId, deadline.Token);
    foreach (var command in new[] { "|", "~", "~0s", ".frame 0", "kn", "dv /t" })
    {
        Log("AFTER PAUSE " + command);
        Log(await transport.ExecuteAsync(command, deadline.Token));
    }
}
catch (Exception exception) { Log(exception.ToString()); }
finally
{
    Log("DISPOSE begin");
    await transport.DisposeAsync();
    transport.ForceTerminateOwnedProcesses();
    Log("DISPOSE end");
}
